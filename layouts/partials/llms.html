{{- /* Publishes /llms.txt at the domain root (https://llmstxt.org/). Called once, from the home page of the default language; the file lists the posts of every language (linking to their Markdown renditions), the projects in site.Params.projects and, under "Optional", the llms-full files published by partials/llms-full.html. */ -}} {{- $lines := slice (printf "# %s" site.Title) "" -}} {{- $lines = $lines | append (printf "> %s" site.Params.llms_summary) "" -}} {{- range hugo.Sites -}} {{- $lines = $lines | append (printf "## Posts (%s)" .Language.Label) "" -}} {{- range where .RegularPages "Section" "posts" -}} {{- $url := .Permalink -}} {{- with .OutputFormats.Get "markdown" -}}{{- $url = .Permalink -}}{{- end -}} {{- $lines = $lines | append (printf "- [%s](%s): %s" .Title $url .Description) -}} {{- end -}} {{- $lines = $lines | append "" -}} {{- end -}} {{- with site.Params.projects -}} {{- $lines = $lines | append "## Projects" "" -}} {{- range . -}} {{- $lines = $lines | append (printf "- [%s](%s): %s" .name .url .description) -}} {{- with .llms -}} {{- $lines = $lines | append (printf "- [%s llms.txt](%s)" $.name .) -}} {{- end -}} {{- end -}} {{- $lines = $lines | append "" -}} {{- end -}} {{- $lines = $lines | append "## Optional" "" -}} {{- range hugo.Sites -}} {{- $name := cond (eq .Language.Lang hugo.Sites.Default.Language.Lang) "llms-full.txt" (printf "llms-full-%s.txt" .Language.Lang) -}} {{- $lines = $lines | append (printf "- [Full text of all posts (%s)](%s)" .Language.Label ($name | absURL)) -}} {{- end -}} {{- (resources.FromString "llms.txt" (delimit ($lines | append "") "\n")).Publish -}}
